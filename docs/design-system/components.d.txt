import type { ReactNode, ButtonHTMLAttributes, InputHTMLAttributes, TextareaHTMLAttributes, MouseEvent } from "react";

export type IconName =
  | "home" | "briefcase" | "flask" | "user" | "layers" | "folder" | "sun" | "moon" | "arrow-up-right" | "arrow-right" | "arrow-left"
  | "star" | "clock" | "calendar" | "git-branch" | "code" | "box" | "sparkle" | "check" | "check-check" | "x" | "alert-circle"
  | "alert-triangle" | "info" | "refresh" | "lock" | "log-out" | "inbox" | "mail" | "send" | "search" | "message" | "compass"
  | "menu" | "cloud-off" | "gauge";

export interface IconProps { name: IconName; size?: number; strokeWidth?: number; label?: string; className?: string }

export interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: "primary" | "secondary" | "ghost" | "brand" | "danger";
  size?: "sm" | "md" | "lg";
  iconLeft?: IconName; iconRight?: IconName;
  /** Desabilita e mostra Spinner — bloqueia envio duplicado. */
  loading?: boolean; loadingText?: string;
  /** Renderiza <a>. */
  href?: string;
}

export interface IconButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  icon: IconName; label: string; variant?: "glass" | "solid" | "ghost"; size?: "sm" | "md" | "lg"; iconSize?: number;
}
